下面给出一套**以《伊卡洛斯消失之谜v0.5.5长视野4》为底座、把《回合制魂游测试v0.2》整套战斗机制移植进来**的替换方案。建议先把 v0.5.5 另存为 `伊卡洛斯消失之谜v0.6回合制.html`，然后按下面 12 个编辑点（E1~E12）依次“定位 → 替换/插入”。所有片段都尽量整段给出，直接覆盖即可。

移植后的总体设计：

- **单一全局状态 `G`**：使用 2D 版的战斗状态结构（`G.entities / G.player / G.turn / G.explored / G.noiseMarkers / G.timeline` 等），伪3D 渲染直接读取 `G.player.pos / G.player.facing`。
- **移动 = 打牌**：自由视角下 `短按W = 朝当前朝向发动主手卡`，`短按A/D = 左右转90°`，`短按S = 转180°`（均不出牌）；锁定朝北模式下 WASD 直接对应世界四个方向出牌（和 2D 版一致）。
- **灯火 = 侦查视野**：玩家 `scout` 由 `LANTERN` 决定（可被明目/失明 BUFF 修正），同时驱动：3D 房间递归渲染深度、迷雾减淡、2D 版 `canSeeCell` 的直线视野。
- **第一人称怪物显示**：正前方直线上、视野内、视线通透的实体，在 3D 画面中央用**竖排名字**表示，距离越近字越大，并附带血条与意图倒计时。
- **小地图**：继续按 2D 迷宫方式显示（生态色块、墙、红色未探通道），并叠加 2D 版的噪音标记与实体影像点。

## E3. 灯火/渲染距离 与 战斗视野挂钩

【定位】

```js
// 【新增】根据灯火计算有效渲染距离
function effectiveRenderDistance() {
return LANTERN > 0 ? LANTERN - 1 : 0;
}
```

【操作】整段替换为：

```js
/* 有效渲染距离：战斗中 = 玩家侦查(灯火基础+BUFF) - 1；未进入战斗时退回灯火参数 */
function effectiveRenderDistance(){
  if(inGame && G && G.player) return Math.max(0, getScout(G.player) - 1);
  return LANTERN > 0 ? LANTERN - 1 : 0;
}
```

```js
/* ============ 几何 / 寻路 ============ */
function cellXY(id){return{x:id%M.n,y:(id/M.n)|0};}
function bfsFrom(src){
  if(!bfsCache.has(src))bfsCache.set(src,bfsFull(M.walls,M.n,M.h,src));
  return bfsCache.get(src);
}
function pathDist(a,b){if(a===b)return 0;return bfsFrom(a).dist[b];}
function lineInfo(a,b){
  const A=cellXY(a),B=cellXY(b);
  if(A.x===B.x){const d=Math.abs(A.y-B.y);return{dist:d,dir:A.y>B.y?0:2};}
  if(A.y===B.y){const d=Math.abs(A.x-B.x);return{dist:d,dir:A.x>B.x?3:1};}
  return null;
}

/* ============ BUFF ============ */
function activeBuffs(e){if(!G||!e.buffs)return[];return e.buffs.filter(b=>b.remaining>0);}
function hasBuff(e,name){return activeBuffs(e).some(b=>b.name===name);}
function sumBuff(e,name){return activeBuffs(e).reduce((s,b)=>b.name===name?s+b.level:0,0);}
function addBuff(e,name,level,dur){if(!e.buffs)e.buffs=[];e.buffs.push({name,level,dur,remaining:Math.max(0,dur)+1});}
function buffText(e){return activeBuffs(e).map(b=>`${b.name}${b.level>0?b.level:''}(${b.remaining-1})`).join(' ');}
function cellHasWall(cell,entrant){
  if(!G)return false;
  for(const e of G.entities){
    if(e.dead||e.pos!==cell)continue;
    const walls=activeBuffs(e).filter(b=>b.name==='盾墙');
    if(walls.length>0&&entrant&&e.faction!==entrant.faction)return true;
  }
  return false;
}
function clearLine(a,b){
  if(a===b)return true;
  const li=lineInfo(a,b);
  if(!li)return false;
  let cur=a;
  for(let i=0;i<li.dist;i++){
    if(M.walls[cur]&BIT[li.dir])return false;
    const x=(cur%M.n)+DIRS[li.dir][0],y=((cur/M.n)|0)+DIRS[li.dir][1];
    if(x<0||y<0||x>=M.n||y>=M.h)return false;
    cur=y*M.n+x;
    if(cur!==b&&cellHasWall(cur,null))return false;
  }
  return true;
}
function canSeeCell(obs,target,scout){
  if(obs===target)return true;
  const li=lineInfo(obs,target);
  if(!li)return false;
  if(li.dist>scout)return false;
  return clearLine(obs,target);
}
function dirBetween(a,b){const li=lineInfo(a,b);return li?li.dir:null;}
function nextDirTowards(from,to){
  if(from===to)return null;
  const r=bfsFrom(to);
  const d=r.dist[from];
  if(d<0)return null;
  const x=from%M.n,y=(from/M.n)|0;
  for(let dir=0;dir<4;dir++){
    if(M.walls[from]&BIT[dir])continue;
    const nx=x+DIRS[dir][0],ny=y+DIRS[dir][1];
    if(nx<0||ny<0||nx>=M.n||ny>=M.h)continue;
    const ni=ny*M.n+nx;
    if(r.dist[ni]===d-1)return dir;
  }
  return null;
}
function randomReachableCell(from){
  for(let i=0;i<80;i++){
    const cell=(Math.random()*M.n*M.h)|0;
    if(pathDist(from,cell)>=0)return cell;
  }
  return M.end;
}
/* 侦查 = 灯火基础 + 明目 - 失明（灯火<=0 时只有 1 格微光） */
function baseScout(){return LANTERN>0?LANTERN:1;}
function getScout(e){
  let s=e.isPlayer?baseScout():(e.scout||0);
  s+=sumBuff(e,'明目');
  s-=sumBuff(e,'失明');
  return Math.max(0,s);
}

function getExecutionOrder(){
  return G.entities.filter(e=>!e.dead).sort((a,b)=>{
    if(b.agi!==a.agi)return b.agi-a.agi;
    const fa=a.faction==='friendly'?0:1,fb=b.faction==='friendly'?0:1;
    if(fa!==fb)return fa-fb;
    const ay=(a.pos/M.n)|0,by=(b.pos/M.n)|0;
    if(ay!==by)return ay-by;
    return(a.pos%M.n)-(b.pos%M.n);
  });
}
function getTaunt(e){
  const pd=G.playerDist?G.playerDist[e.pos]:999;
  const d=pd<0?999:pd;
  return -128*d + e.appearance + (e.extraTaunt||0);
}
function isTargetable(user,target,mode='attack'){
  if(target===user||target.dead)return false;
  if(mode==='support'){
    if(user.faction==='friendly')return target.faction!=='enemy';
    if(user.faction==='enemy')return target.faction!=='friendly';
    return target.faction!==user.faction;
  }
  if(user.faction==='friendly')return target.faction!=='friendly';
  if(user.faction==='enemy')return target.faction!=='enemy';
  return target.faction!==user.faction;
}
function selectTargets(user,card,dir,mode='attack'){
  const cand=new Map();
  const addAt=cell=>{for(const e of G.entities){if(e.pos===cell&&isTargetable(user,e,mode))cand.set(e.id,e);}};
  if(card.rangePath!=null){
    const rpMin=Array.isArray(card.rangePath)?card.rangePath[0]:0;
    const rpMax=Array.isArray(card.rangePath)?card.rangePath[1]:card.rangePath;
    const r=bfsFrom(user.pos);
    for(const e of G.entities){
      if(!isTargetable(user,e,mode))continue;
      const d=r.dist[e.pos];
      if(d>=rpMin&&d<=rpMax)cand.set(e.id,e);
    }
  }else{
    const range=card.range||[0,0];
    const min=range[0]??0,max=range[1]??0;
    if(min===0)addAt(user.pos);
    if(max>0){
      const dirs=dir==null?[0,1,2,3]:[dir];
      for(const d of dirs){
        let cur=user.pos;
        for(let step=1;step<=Math.min(max,200);step++){
          if(M.walls[cur]&BIT[d])break;
          const x=(cur%M.n)+DIRS[d][0],y=((cur/M.n)|0)+DIRS[d][1];
          if(x<0||y<0||x>=M.n||y>=M.h)break;
          cur=y*M.n+x;
          if(step>=min)addAt(cur);
        }
      }
    }
  }
  const arr=[...cand.values()].sort((a,b)=>getTaunt(b)-getTaunt(a));
  return arr.slice(0,card.multi||1);
}
function canHitLine(user,card,target){
  const range=card.range||[0,0];
  const min=range[0]??0,max=range[1]??0;
  if(min===0&&user.pos===target.pos)return true;
  if(max<=0)return false;
  const li=lineInfo(user.pos,target.pos);
  if(!li)return false;
  if(li.dist<min||li.dist>max)return false;
  return clearLine(user.pos,target.pos);
}
function isAttackLike(card){

}
function getIntentCells(user,card,dir){
  if(!isAttackLike(card))return[];
  const cells=new Set();
  if(card.rangePath!=null){
    const rpMin=Array.isArray(card.rangePath)?card.rangePath[0]:0;
    const rpMax=Array.isArray(card.rangePath)?card.rangePath[1]:card.rangePath;
    const r=bfsFrom(user.pos);
    for(let i=0;i<r.dist.length;i++){if(r.dist[i]>=rpMin&&r.dist[i]<=rpMax)cells.add(i);}
    return[...cells];
  }
  const range=card.range||[0,0];
  const min=range[0]??0,max=range[1]??0;
  if(min===0)cells.add(user.pos);
  if(max>0){
    const dirs=dir==null?[0,1,2,3]:[dir];
    for(const d of dirs){
      let cur=user.pos;
      for(let step=1;step<=Math.min(max,200);step++){
        if(M.walls[cur]&BIT[d])break;
        const x=(cur%M.n)+DIRS[d][0],y=((cur/M.n)|0)+DIRS[d][1];
        if(x<0||y<0||x>=M.n||y>=M.h)break;
        cur=y*M.n+x;
        if(step>=min)cells.add(cur);
      }
    }
  }
  return[...cells];
}

function updateNoiseMarkers(source,r,short,dist){
  const p=G.player;
  if(dist<=0)return;
  const entries=[];
  const px=p.pos%M.n,py=(p.pos/M.n)|0;
  for(let d=0;d<4;d++){
    const nx=px+DIRS[d][0],ny=py+DIRS[d][1];
    if(nx<0||ny<0||nx>=M.n||ny>=M.h)continue;
    const ni=ny*M.n+nx;
    if(r.dist[ni]===dist-1){
      const dirToPlayer=OPP[d];
      entries.push({cell:ni,arrow:ARROWS[dirToPlayer],dir:dirToPlayer});
    }
  }
  if(entries.length)G.noiseMarkers.set(source.id,{entries,short,dist,turn:G.turn});
}
function emitNoise(source,level,label,short){
  if(level<=0||source.dead)return;
  if(source.permanentSilent||source.silentTurns>0)return;
  const r=bfsFrom(source.pos);
  for(const obs of G.entities){
    if(obs===source||obs.dead)continue;
    if(obs.faction===source.faction)continue;
    if(obs.isPlayer){
      const hear=obs.hearing||0;
      const d=r.dist[obs.pos];
      if(d>=0&&d<=level+hear){
        obs.noises.set(source.id,{cell:source.pos,turn:G.turn,label,short,dist:d});
        updateNoiseMarkers(source,r,short,d);
      }
    }else if(obs.faction==='enemy'){
      if(source.faction!=='friendly')continue;
      const hear=obs.hearing||0;
      const d=r.dist[obs.pos];
      if(d>=0&&d<=level+hear){
        obs.noises.set(source.id,{cell:source.pos,turn:G.turn,label,short,dist:d});
      }
    }
  }
}
function refreshPlayerImages(){
  const p=G.player;
  p.images.clear();
  const sc=getScout(p);
  for(const e of G.entities){
    if(e===p||e.dead)continue;
    if(canSeeCell(p.pos,e.pos,sc)){
      p.images.set(e.id,{cell:e.pos,turn:G.turn,name:e.name});
    }
  }
  checkSeenExecutioner();
}
function emitEndImages(){
  for(const src of G.entities){
    if(src.dead)continue;
    for(const obs of G.entities){
      if(obs===src||obs.dead)continue;
      if(obs.faction===src.faction)continue;
      if(obs.isPlayer&&src.faction==='friendly')continue;
      if(obs.faction==='enemy'&&src.faction!=='friendly')continue;
      if(canSeeCell(obs.pos,src.pos,getScout(obs))){
        obs.images.set(src.id,{cell:src.pos,turn:G.turn,name:src.name});
      }
    }
  }
  refreshPlayerImages();
}

/* ============ 伤害 / 死亡 / 奖励 ============ */
function dealDamage(source,target,dmg,card){
  if(target.dead)return;
  let d=dmg||0;
  const melee=!card||!card.ranged;
  if(melee&&hasBuff(target,'近战免疫'))d=0;
  if(!melee&&hasBuff(target,'远程免疫'))d=0;
  if(melee&&hasBuff(target,'闪避')&&Math.random()<0.5){
    if(target.isPlayer)showSubtitle('【闪避】');
    return;
  }
  if(source){d+=sumBuff(source,'力量');d-=sumBuff(source,'虚弱');}
  d+=sumBuff(target,'脆弱');d-=sumBuff(target,'防御');
  if(d<0)d=0;
  const barrier=activeBuffs(target).filter(b=>b.name==='屏障');
  for(const b of barrier){
    if(d<=0)break;
    const absorb=Math.min(b.level,d);
    b.level-=absorb;
    d-=absorb;
  }
  if(target.buffs)target.buffs=target.buffs.filter(b=>!(b.name==='屏障'&&b.level<=0));
  if(d>0){
    target.hp-=d;
    if(target.isPlayer)flashScreen('hurt');
    if(target.hp<=0)killEntity(target,source);
  }
}
function killEntity(e,source){
  if(e.dead)return;
  e.dead=true;e.hp=0;e.intent=null;
  if(e.isPlayer){showEnd(false);}
  else{
    if(e.type==='executioner')showEnd(true);
    if(e.type==='interrogator'){
      grantCards('pistol',3);
      const ex=getExecutioner();
      if(ex){
        e.images.forEach((v,k)=>ex.images.set(k,{...v,turn:G.turn}));
        e.noises.forEach((v,k)=>ex.noises.set(k,{...v,turn:G.turn}));
        ex.hp=Math.max(0,Math.floor(ex.hp*0.5));
        toast('克里特拷问官死亡，行刑官损失了50%生命');
        if(ex.hp<=0)killEntity(ex,e);
      }
    }
    if(e.type==='barrel')grantRandomCards(1,NONBODY_PLAYER_CARDS);
    if(e.type==='blood')grantRandomCards(3,ALL_PLAYER_CARDS);
    if(e.type==='armed'){grantCard('riotshield');grantRandomCards(2,NONBODY_PLAYER_CARDS);}
    if(e.type==='berserk'){grantCard('grinder');grantRandomCards(2,NONBODY_PLAYER_CARDS);}
  }
  updateUI();
}
function grantCard(id,silent){
  const c=CARD_DB[id];
  if(!c||!G||!G.player)return;
  if(!G.player.cards[id])G.player.cards[id]={uses:0};
  const pc=G.player.cards[id];
  if(pc.uses!==Infinity)pc.uses+=(c.uses===Infinity?1:c.uses);
  if(!silent)toast(`获得 ${c.name}`);
  if(openCat)renderModal();
}
function grantCards(id,n){
  for(let i=0;i<n;i++)grantCard(id,true);
  toast(`获得 ${CARD_DB[id].name} ×${n}`);
  if(openCat)renderModal();
}
function grantRandomCards(n,pool){
  for(let i=0;i<n;i++){
    const id=pool[(Math.random()*pool.length)|0];
    grantCard(id,false);
  }
}

  toast(`打断了 ${t.name}`);
}
function doMove(e,dir,card){
  if(dir==null)return;
  if(e.isPlayer)e.facing=dir;
  let steps=card.move||1;
  for(let i=0;i<steps;i++){
    if(M.walls[e.pos]&BIT[dir])break;
    const nx=(e.pos%M.n)+DIRS[dir][0],ny=((e.pos/M.n)|0)+DIRS[dir][1];
    if(nx<0||ny<0||nx>=M.n||ny>=M.h)break;
    const next=ny*M.n+nx;
    if(cellHasWall(next,e))break;
    assignCell(e,next);
    if(e.isPlayer){
      e.steps++;
      G.moveFx=performance.now();
      G.lastMoveDir=dir;
      if(footstepEngine)footstepEngine.playBiomeStep(BIOMES[M.biome[next]].id);
      if(card.id==='slide'&&i===0)interruptFirstCell(e.pos);
    }
  }
  if(e.isPlayer)revealVision();
  if(card.noise)emitNoise(e,card.noise,card.name,card.short||card.name[0]);
}
function performAttack(user,card,dir){
  const targets=selectTargets(user,card,dir,'attack');
  if(card.noise)emitNoise(user,card.noise,card.name,card.short||'响');
  for(const t of targets){
    const hits=card.hits||1;
    for(let h=0;h<hits;h++){
      if(t.dead)break;
      dealDamage(user,t,card.dmg,card);
    }
    if(card.bleed&&!t.dead)addBuff(t,'流血',card.bleed.level,card.bleed.dur);
    if(card.debuff&&!t.dead)addBuff(t,card.debuff.name,card.debuff.level,card.debuff.dur);
    if(card.staminaDamage&&t.isPlayer){
      t.stamina=Math.max(0,t.stamina-card.staminaDamage);
      flashScreen('stamina');
    }
  }
}
function doCharge(user,card,dir){
  if(dir==null)dir=0;
  if(user.isPlayer)user.facing=dir;
  const max=card.range?card.range[1]:10;
  for(let i=0;i<max;i++){
    if(M.walls[user.pos]&BIT[dir])break;
    const nx=(user.pos%M.n)+DIRS[dir][0],ny=((user.pos/M.n)|0)+DIRS[dir][1];
    if(nx<0||ny<0||nx>=M.n||ny>=M.h)break;
    const next=ny*M.n+nx;
    if(cellHasWall(next,user))break;
    assignCell(user,next);
    if(user.isPlayer){user.steps++;G.moveFx=performance.now();G.lastMoveDir=dir;}
    if(G.entities.some(e=>!e.dead&&e!==user&&e.pos===next))break;
  }
  if(user.isPlayer)revealVision();
  const impact={dmg:15,range:[0,0],multi:3,melee:true};
  const targets=selectTargets(user,impact,null,'attack');
  for(const t of targets)dealDamage(user,t,15,impact);
  emitNoise(user,4,'血腥撞击','撞');
}
function doBlock(user){
  if(!user.isPlayer)return;
  if(G.blockUsedTurn===G.turn)return;
  const offId=user.offhand;
  if(!offId)return;
  const offC=CARD_DB[offId];
  if(!offC||offC.cat!=='weapon')return;
  const pc=user.cards[offId];
  if(!pc||pc.uses<=0)return;
  pc.uses--;
  G.blockUsedTurn=G.turn;
  const atk=getCardAttack(offC);
  addBuff(user,'屏障',atk,1);
  toast(`获得 屏障${atk}`);
}
function doCallTerror(user){
  const ex=getExecutioner();
  if(!ex)return;
  user.images.forEach((v,k)=>ex.images.set(k,{...v,turn:G.turn}));
  user.noises.forEach((v,k)=>ex.noises.set(k,{...v,turn:G.turn}));
  for(let i=0;i<2;i++){
    let dir=null;
    if(ex.lastTargetCell!=null)dir=nextDirTowards(ex.pos,ex.lastTargetCell);
    if(dir==null){
      const opts=[];
      for(let d=0;d<4;d++)if(!(M.walls[ex.pos]&BIT[d]))opts.push(d);
      dir=opts.length?opts[(Math.random()*opts.length)|0]:null;
    }
    if(dir!=null){
      const nx=(ex.pos%M.n)+DIRS[dir][0],ny=((ex.pos/M.n)|0)+DIRS[dir][1];
      if(nx>=0&&ny>=0&&nx<M.n&&ny<M.h)assignCell(ex,ny*M.n+nx);
    }
  }
  toast('呼唤恐怖！');
}

function doIaido(user,dir){
  if(dir==null)dir=user.facing||0;
  if(user.isPlayer)user.facing=dir;
  let firstCell=user.pos;
  for(let i=0;i<2;i++){
    if(M.walls[user.pos]&BIT[dir])break;
    const nx=(user.pos%M.n)+DIRS[dir][0],ny=((user.pos/M.n)|0)+DIRS[dir][1];
    if(nx<0||ny<0||nx>=M.n||ny>=M.h)break;
    const next=ny*M.n+nx;
    if(cellHasWall(next,user))break;
    if(i===0)firstCell=next;
    assignCell(user,next);
    if(user.isPlayer){user.steps++;G.moveFx=performance.now();G.lastMoveDir=dir;}
  }
  if(user.isPlayer)revealVision();
  const target=G.entities.find(e=>!e.dead&&e!==user&&e.pos===firstCell&&isTargetable(user,e,'attack'));
  if(target&&user.isPlayer){
    const offId=user.offhand;
    if(offId==='katana'){
      const pc=user.cards[offId];
      if(pc){
        while(pc.uses>0){pc.uses--;dealDamage(user,target,10,CARD_DB.katana);}
        toast('拔刀斩！');
      }
    }
  }
}
function spawnPlace(pos,type){
  if(type==='cover'){addEntity({name:'木制掩体',type:'cover',faction:'neutral',pos,hp:15,agi:0,permanentSilent:true,extraTaunt:10});toast('放置木制掩体');}
  else if(type==='selftrap'){addEntity({name:'自制陷阱',type:'selftrap',faction:'friendly',pos,hp:5,agi:0});toast('放置自制陷阱');}

async function executeCard(user,cardId,dir,fx=true){
  const card=CARD_DB[cardId];
  if(!card)return;
  if(dir!=null&&user.isPlayer)user.facing=dir;
  if(fx)actionFx(user,card);
  if(card.iaido){doIaido(user,dir);return;}
  if(card.move&&!card.iaido){doMove(user,dir,card);return;}
  switch(card.id){
    case 'block':doBlock(user);return;
    case 'callTerror':doCallTerror(user);return;
    case 'callBigTerror':doCallBigTerror(user);return;
    case 'charge':case 'exeCharge':doCharge(user,card,dir);return;
    case 'lampoil':case 'dodge':case 'chocolate':
      if(card.buff)addBuff(user,card.buff.name,card.buff.level,card.buff.dur);return;
    case 'woodshield':
      if(card.shieldBuff)addBuff(user,card.shieldBuff.name,card.shieldBuff.level,card.shieldBuff.dur);return;
    case 'riotshield':
      if(card.shieldBuff)addBuff(user,card.shieldBuff.name,card.shieldBuff.level,card.shieldBuff.dur);
      if(card.wallBuff)addBuff(user,card.wallBuff.name,card.wallBuff.level,card.wallBuff.dur);return;
    case 'suture':case 'bandage':case 'redvial':case 'redflask':case 'drink':
      user.hp=Math.min(user.maxhp,user.hp+card.heal);
      if(user.isPlayer)toast(`回复 ${card.heal}`);
      if(card.removeBleed&&user.buffs)user.buffs=user.buffs.filter(b=>b.name!=='流血');
      if(card.grant&&user.isPlayer)grantCard(card.grant);return;
    case 'bluevial':case 'blueflask':
      if(user.isPlayer){
        user.stamina=Math.min(user.maxStamina,user.stamina+card.staminaHeal);
        flashScreen('stamina');
        if(card.grant)grantCard(card.grant);
      }return;
    case 'warcry':emitNoise(user,card.noise,card.name,'吼');return;
    case 'music':emitNoise(user,card.noise,card.name,'乐');return;
    case 'coverItem':spawnPlace(user.pos,'cover');return;
    case 'trapItem':spawnPlace(user.pos,'selftrap');return;
    case 'musicboxItem':spawnPlace(user.pos,'musicbox');return;
  }
  if(isAttackLike(card))performAttack(user,card,dir);
}
function consumeEnemyCard(e,cardId){
  if(e.weapons&&Array.isArray(e.weapons)){const i=e.weapons.indexOf(cardId);if(i>=0)e.weapons.splice(i,1);}
  if(e.abilities&&Array.isArray(e.abilities)){const i=e.abilities.indexOf(cardId);if(i>=0)e.abilities.splice(i,1);}
}

  if(!e.intent)return;
  const it=e.intent;
  e.intent=null;
  await executeCard(e,it.cardId,it.dir,true);
  if(e.faction==='enemy'&&(e.type==='executioner'||e.type==='interrogator'))consumeEnemyCard(e,it.cardId);
}

function freshInfos(map){
  const arr=[];
  map.forEach((info,srcId)=>{
    const src=getEntity(srcId);
    if(src&&!src.dead&&info.turn>G.turn-10)arr.push({srcId,info});
  });
  return arr;
}
function chooseInfoTarget(e,arr){
  const r=bfsFrom(e.pos);
  let best=null;
  for(const it of arr){
    const src=getEntity(it.srcId);
    if(!src||src.dead)continue;
    const d=r.dist[it.info.cell];
    if(d<0)continue;
    if(!best||d<best.d||(d===best.d&&it.info.turn>best.turn))best={d,turn:it.info.turn,cell:it.info.cell,srcId:it.srcId};
  }
  return best;
}
function getAttackOptions(e){
  if(e.type==='interrogator')return['pistol',...(e.abilities||[])];
  if(e.type==='executioner')return[...(e.weapons||[]),...(e.abilities||[])].filter(id=>id!=='drink');
  if(e.type==='blood')return['shovel','bat','spear'];
  if(e.type==='armed')return['shovel','bat','spear'];
  if(e.type==='berserk')return['grinder'];
  return[];
}
function canEnemyUse(e,cid,source,hasImage){
  const card=CARD_DB[cid];
  if(!card)return false;
  if(cid==='drink')return e.hp<e.maxhp;
  if(cid==='callTerror'||cid==='callBigTerror'){
    return !!getExecutioner()&&(freshInfos(e.images).length||freshInfos(e.noises).length);
  }
  if(cid==='senseDeprive'||cid==='sleepDeprive'){
    return !!source&&pathDist(e.pos,source.pos)>=0&&pathDist(e.pos,source.pos)<=2;
  }
  if(card.rangePath!=null){
    const rpMax=Array.isArray(card.rangePath)?card.rangePath[1]:card.rangePath;
    return !!source&&pathDist(e.pos,source.pos)>=0&&pathDist(e.pos,source.pos)<=rpMax;
  }
  if(!source||!canHitLine(e,card,source))return false;
  if(card.range&&card.range[1]>0&&e.pos!==source.pos&&dirBetween(e.pos,source.pos)==null)return false;
  if(!hasImage&&e.pos!==source.pos)return false;
  return true;
}
function updateExecutionerCards(e){
  if(G.turn>0&&G.turn%4===0&&(e.weapons||[]).length<3){
    const ws=['axe','mace','flail'];
    e.weapons.push(ws[(Math.random()*ws.length)|0]);
  }
  if(G.turn>0&&G.turn%10===0&&(e.abilities||[]).length<2){
    const as=['exeCharge','drink'];
    e.abilities.push(as[(Math.random()*as.length)|0]);
  }
}
function updateInterrogatorCards(e){
  if(G.turn>0&&G.turn%5===0&&(e.abilities||[]).length<2){
    const as=['callTerror','callBigTerror','senseDeprive','sleepDeprive'];
    e.abilities.push(as[(Math.random()*as.length)|0]);
  }
}
function chooseIntent(e){
  if(e.dead)return;
  if(e.type==='musicbox'){setIntent(e,'music',null);return;}
  if(e.type==='selftrap'){setIntent(e,'spikes',null);return;}
  if(e.type==='spikeTrap'){setIntent(e,'spikes',null);return;}
  if(e.type==='cover'||e.type==='barrel')return;
  if(e.type==='executioner')updateExecutionerCards(e);
  if(e.type==='interrogator')updateInterrogatorCards(e);
  const imgs=freshInfos(e.images);
  const noises=freshInfos(e.noises);
  let target=null,hasImage=false;
  if(imgs.length){target=chooseInfoTarget(e,imgs);hasImage=true;}
  if(!target&&noises.length)target=chooseInfoTarget(e,noises);
  let source=null;
  if(target){
    source=getEntity(target.srcId);
    if(source&&!source.dead){
      e.combat=true;
      e.lastInfoTurn=target.turn;
      e.lastTargetCell=hasImage?source.pos:target.cell;
    }
  }
  if(!target&&e.combat&&G.turn-(e.lastInfoTurn||0)>=10)e.combat=false;
  if(e.combat&&source&&!source.dead){
    if(e.shieldType&&Math.random()<e.defChance){setIntent(e,e.shieldType,null);return;}
    if(e.type==='executioner'&&(e.abilities||[]).includes('drink')&&e.hp<=e.maxhp*.5){setIntent(e,'drink',null);return;}
    const options=getAttackOptions(e).filter(id=>id!=='drink');
    const usable=options.filter(cid=>canEnemyUse(e,cid,source,hasImage));
    if(usable.length){
      const cid=usable[(Math.random()*usable.length)|0];
      const card=CARD_DB[cid];
      let dir=null;
      if(card.range&&card.range[1]>0)dir=dirBetween(e.pos,source.pos);
      setIntent(e,cid,dir);
      return;
    }
    const dir=nextDirTowards(e.pos,e.lastTargetCell);
    if(dir!=null){setIntent(e,'sprint',dir);return;}
  }
  if(e.type==='executioner'){
    const inter=getInterrogator();
    if(inter){
      e.lastTargetCell=inter.pos;
      const dir=nextDirTowards(e.pos,inter.pos);
      if(dir!=null){setIntent(e,'walk',dir);return;}
    }
  }
  if(e.patrolTarget==null||e.pos===e.patrolTarget||pathDist(e.pos,e.patrolTarget)<0){
    e.patrolTarget=randomReachableCell(e.pos);
  }
  e.lastTargetCell=e.patrolTarget;
  const dir=nextDirTowards(e.pos,e.patrolTarget);
  if(dir!=null)setIntent(e,'walk',dir);
  else e.patrolTarget=null;
}

function processBuffMachine(){
  for(const e of G.entities){
    if(e.dead)continue;
    const bleeds=activeBuffs(e).filter(b=>b.name==='流血');
    if(bleeds.length){
      const dmg=bleeds.reduce((s,b)=>s+b.level,0);
      e.hp-=dmg;
      if(e.isPlayer)flashScreen('hurt');
      if(e.hp<=0)killEntity(e,null);
    }
  }
}
function spawnPrisonerIfLow(){
  const cnt=G.entities.filter(e=>!e.dead&&(e.type==='berserk'||e.type==='blood'||e.type==='armed')).length;
  if(cnt>=4)return;
  const corners=[0,M.n-1,(M.h-1)*M.n,M.h*M.n-1];
  const pd=bfsFrom(G.player.pos).dist;
  let best=corners[0],bd=-1;
  for(const c of corners){const d=pd[c];if(d>bd){bd=d;best=c;}}
  const types=['berserk','blood','armed'];
  addPrisoner(types[(Math.random()*types.length)|0],best);
  toast('角落传来铁链声：新的囚犯出现了');
}
async function advanceTurn(){
  if(!G||G.over)return;
  G.turn++;
  processBuffMachine();
  if(G.over)return;
  const order=getExecutionOrder();
  for(const e of order){
    if(e.dead)continue;
    if(e.intent&&e.intent.executeTurn<=G.turn){
      await executeIntent(e);
      if(G.over)return;
    }
  }
  const order2=getExecutionOrder();
  for(const e of order2){
    if(e.dead||e.isPlayer)continue;
    if(e.permanentSilent||e.silentTurns>0)continue;
    if(e.intent)continue;
    chooseIntent(e);
  }
  for(const e of G.entities){
    if(e.dead)continue;
    if(e.silentTurns>0)e.silentTurns--;
    if(e.isPlayer&&e.stamina<e.maxStamina)e.stamina++;
  }
  for(const e of G.entities){
    if(e.dead||!e.buffs)continue;
    for(const b of e.buffs)b.remaining--;
    e.buffs=e.buffs.filter(b=>b.remaining>0);
  }
  if(G.turn%30===0)spawnPrisonerIfLow();
  emitEndImages();
  revealVision();
  updateUI();
  await wait(25);
}

async function playerPlayCard(cardId,dir){
  if(!inGame||!G||G.over||G.busy)return;
  if(!cardId){toast('主手为空');sfx.deny();return;}
  if(G.player.intent){toast('行为尚未完成');sfx.deny();return;}
  const card=CARD_DB[cardId];
  const pc=G.player.cards[cardId];
  if(!pc){toast('未持有该行为');sfx.deny();return;}
  if(pc.uses!==Infinity&&pc.uses<=0){toast('次数不足');sfx.deny();return;}
  if(card.stamina&&G.player.stamina<card.stamina){toast('体力不足');sfx.deny();return;}
  if(card.id==='block'){
    if(G.blockUsedTurn===G.turn){toast('本回合已格挡');sfx.deny();return;}
    const off=G.player.offhand;
    if(!off){toast('副手需要武器');sfx.deny();return;}
    const offC=CARD_DB[off];
    if(!offC||offC.cat!=='weapon'){toast('副手需要武器');sfx.deny();return;}
    const offPc=G.player.cards[off];
    if(!offPc||offPc.uses<=0){toast('副手武器次数不足');sfx.deny();return;}
  }
  if(card.stamina){G.player.stamina-=card.stamina;flashScreen('stamina');}
  if(pc.uses!==Infinity)pc.uses--;
  if(dir!=null)G.player.facing=dir;
  G.busy=true;
  if(card.cost===0){
    await executeCard(G.player,cardId,dir,true);
    await wait(80);
  }else{
    setIntent(G.player,cardId,dir);
    for(let i=0;i<card.cost;i++){
      if(G.over)break;
      await wait(420);
      await advanceTurn();
    }
  }
  G.busy=false;
  updateUI();
}

function randCellNotNearPlayer(){
  const sx=M.start%M.n,sy=(M.start/M.n)|0;
  const startDist=bfsFrom(M.start).dist;
  for(let i=0;i<1000;i++){
    const cell=(Math.random()*M.n*M.h)|0;

    if(cell===M.start||cell===M.end)continue;
    if(startDist[cell]>=0)return cell;
  }
  return M.end;
}
```

```js
function renderMap(now){
  const mul=mapSizeMultiplier;
  const SZ=170*mul;
  mmg.clearRect(0,0,SZ,SZ);
  if(!G||!inGame||!G.player)return;
  /* 旋转：自由视角跟随玩家朝向；锁定朝北固定 */
  const targetRot=viewMode==='north'?0:worldDeg(G.player.facing);
  rotC+=((targetRot-rotC+540)%360-180)*.22;
  const phi=-(rotC+90)*Math.PI/180,co=Math.cos(phi),si=Math.sin(phi);
  const R=MZL[mzi],scale=Math.max(1,Math.floor(SZ/(R*8))),cs=scale*8;
  const cx=SZ/2,cy=SZ/2,px=G.player.pos%M.n,py=(G.player.pos/M.n)|0;
  const rot=(dx,dy)=>[dx*co-dy*si,dx*si+dy*co];
  /* 1. 彩色房间（已探明） */
  for(const id of G.explored){
    const dx=id%M.n-px,dy=(id/M.n|0)-py;
    if(Math.abs(dx)>R||Math.abs(dy)>R)continue;
    const[rx,ry]=rot(dx,dy),sx=cx+rx*cs,sy=cy+ry*cs;
    if(sx<-cs||sy<-cs||sx>SZ+cs||sy>SZ+cs)continue;
    const cur=id===G.player.pos;
    const qc=quadColors(BIOMES[M.biome[id]].id);
    if(qc){
      const qd=[[-.5,-.5,0],[0,-.5,1],[-.5,0,2],[0,0,3]];
      for(const[qx,qy,qi]of qd)mmPoly(mmg,mmCellCorners(sx,sy,cs,co,si,qx,qy),cur?mixc(qc[qi],'#ffffff',.35):qc[qi]);
    }else{
      const corners=[[-.5,-.5],[.5,-.5],[.5,.5],[-.5,.5]].map(p=>[sx+(p[0]*co-p[1]*si)*cs,sy+(p[0]*si+p[1]*co)*cs]);
      mmPoly(mmg,corners,cur?mixc(BIOMES[M.biome[id]].prev,'#ffffff',.35):BIOMES[M.biome[id]].prev);
    }
    /* 当前视野外变暗 */
    if(!G.renderCells.has(id)){
      mmg.fillStyle='rgba(0,0,0,.45)';
      const corners=[[-.5,-.5],[.5,-.5],[.5,.5],[-.5,.5]].map(p=>[sx+(p[0]*co-p[1]*si)*cs,sy+(p[0]*si+p[1]*co)*cs]);
      mmPoly(mmg,corners,'rgba(0,0,0,.45)');
    }
  }
  /* 2. 挖墙 */
  mmg.globalCompositeOperation='destination-out';
  for(const id of G.explored){
    const dx=id%M.n-px,dy=(id/M.n|0)-py;
    if(Math.abs(dx)>R||Math.abs(dy)>R)continue;
    const[rx,ry]=rot(dx,dy),sx=cx+rx*cs,sy=cy+ry*cs;
    if(sx<-cs||sy<-cs||sx>SZ+cs||sy>SZ+cs)continue;
    const w=M.walls[id];
    for(let d=0;d<4;d++){
      if(w&BIT[d]){mmStrip(mmg,sx,sy,cs,d,0,8,'#000',co,si);continue;}
      const wide=wallVariant(id,d)===2;
      if(wide){mmStrip(mmg,sx,sy,cs,d,0,1,'#000',co,si);mmStrip(mmg,sx,sy,cs,d,7,8,'#000',co,si);}
      else{mmStrip(mmg,sx,sy,cs,d,0,3,'#000',co,si);mmStrip(mmg,sx,sy,cs,d,5,8,'#000',co,si);}
    }
  }
  mmg.globalCompositeOperation='source-over';
  /* 3. 红色未探明通道 */
  for(const id of G.explored){
    const dx=id%M.n-px,dy=(id/M.n|0)-py;
    if(Math.abs(dx)>R||Math.abs(dy)>R)continue;
    const[rx,ry]=rot(dx,dy),sx=cx+rx*cs,sy=cy+ry*cs;
    if(sx<-cs||sy<-cs||sx>SZ+cs||sy>SZ+cs)continue;
    const w=M.walls[id];
    for(let d=0;d<4;d++){
      if(w&BIT[d])continue;
      const ni=((id/M.n|0)+DIRS[d][1])*M.n+(id%M.n)+DIRS[d][0];
      if(G.explored.has(ni))continue;
      const wide=wallVariant(id,d)===2;
      if(wide)mmStrip(mmg,sx,sy,cs,d,1,7,'#e23b2e',co,si);
      else mmStrip(mmg,sx,sy,cs,d,3,5,'#e23b2e',co,si);
    }
  }
  /* 4. 实体影像（玩家当前视野内的） */
  G.player.images.forEach((img,id)=>{
    const e=getEntity(id);
    if(!e||e.dead||img.turn<G.turn)return;
    const dx=img.cell%M.n-px,dy=(img.cell/M.n|0)-py;
    if(Math.abs(dx)>R||Math.abs(dy)>R)return;
    const[rx,ry]=rot(dx,dy),sx=cx+rx*cs,sy=cy+ry*cs;
    mmg.fillStyle=e.faction==='enemy'?'#ff5a4a':(e.faction==='friendly'?'#7ee08a':'#ffd76a');
    mmg.fillRect(sx-3*mul,sy-3*mul,6*mul,6*mul);
    mmg.font=`${10*mul}px monospace`;
    mmg.textAlign='center';mmg.textBaseline='middle';
    mmg.fillStyle='#000';mmg.fillText(e.name[0],sx+1,sy+1);
    mmg.fillStyle='#fff';mmg.fillText(e.name[0],sx,sy);
  });
  /* 5. 噪音标记 */
  G.noiseMarkers.forEach(m=>{
    if(G.turn-m.turn>10)return;
    m.entries.forEach(en=>{
      const dx=en.cell%M.n-px,dy=(en.cell/M.n|0)-py;
      if(Math.abs(dx)>R||Math.abs(dy)>R)return;
      const[rx,ry]=rot(dx,dy),sx=cx+rx*cs,sy=cy+ry*cs;
      const txt=`${en.arrow}${m.short}${m.dist}`;
      mmg.font=`${11*mul}px monospace`;
      mmg.textAlign='center';mmg.textBaseline='middle';
      mmg.lineWidth=2*mul;mmg.strokeStyle='#000';
      mmg.strokeText(txt,sx,sy);
      mmg.fillStyle='#ffb648';
      mmg.fillText(txt,sx,sy);
    });
  });
  /* 6. 玩家箭头 + N */
  mmg.fillStyle='#ffb648';mmg.strokeStyle='#201405';mmg.lineWidth=2*mul;
  mmg.beginPath();
  mmg.moveTo(cx,cy-6*mul);
  mmg.lineTo(cx+5*mul,cy+5*mul);
  mmg.lineTo(cx-5*mul,cy+5*mul);
  mmg.closePath();mmg.fill();mmg.stroke();
  const[nx,ny]=rot(0,-1);
  const fontSize=36*mul,nOffset=66*mul,nYOffset=12*mul;
  mmg.font=`${fontSize}px "VT323",monospace`;
  mmg.textAlign='center';
  mmg.lineWidth=2*mul;mmg.strokeStyle='#000';
  mmg.strokeText('N',cx+nx*nOffset,cy+ny*nOffset+nYOffset);
  mmg.fillStyle='#ffffff';
  mmg.fillText('N',cx+nx*nOffset,cy+ny*nOffset+nYOffset);
}
```

## E11. 大地图 / 人偶 / `curFacing` 适配

【操作】三处小替换：

1. 【定位】

```js
function curFacing(){return(viewMode==='north'||!G||!G.player)?0:G.player.facing;}
```

```js
paintUnits(g,u,0,0,M.n,0,0,G.visited);
```

替换为：

```js
paintUnits(g,u,0,0,M.n,0,0,G.explored);
```

同函数内：

```js
const x=G.pos%M.n*cellSize+cellSize/2,y=(G.pos/M.n|0)*cellSize+cellSize/2;
```

替换为：

```js
const x=G.player.pos%M.n*cellSize+cellSize/2,y=(G.player.pos/M.n|0)*cellSize+cellSize/2;
```

3. 【定位】`function renderDoll(now){ ... }` 整个函数，替换为（用移动脉冲决定走/站姿势）：

```js
function renderDoll(now){
  if(!inGame||!G||!G.player||!$('setDoll').checked)return;
  const moving=(now-(G.moveFx||0))<300;
  const positions=[
    {x:DOLL_X,y:DOLL_Y,s:DOLL_S},
    {x:LEFT_DOLL_X,y:LEFT_DOLL_Y,s:DOLL_S},
    {x:RIGHT_DOLL_X,y:RIGHT_DOLL_Y,s:DOLL_S}
  ];
  for(let i=0;i<3;i++){
    const modelId=dollSlots[i];
    if(!modelId||!Dolls[modelId])continue;
    const Doll=Dolls[modelId];
    const pos=positions[i];
    const dir=(i===0)?G.player.facing:(G.lastMoveDir||G.player.facing);
    const vd=viewDirOf(dir);
    const pose=Doll.poseFor(vd,moving&&i===0);
    vctx.save();
    Doll.draw(vctx,pose,pos.x,pos.y,pos.s,now);
    vctx.restore();
  }
}
```
